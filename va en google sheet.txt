function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "No data received"}))
                    .setMimeType(ContentService.MimeType.JSON);
    }
    
    var data = JSON.parse(e.postData.contents);
    var action = data.action || "create";
    var lastRow = sheet.getLastRow();
    
    // Si hay datos, leemos los valores de la columna ID (Columna A / índice 1) de golpe
    var ids = lastRow > 1 ? sheet.getRange(2, 1, lastRow - 1, 1).getValues() : [];
    
    if (action === "create") {
      sheet.appendRow([
        data.id || Date.now(),
        data.fecha || new Date().toISOString(),
        data.nombre || "",
        data.producto || "Mango Con Crema 267 ml",
        data.cantidad || 0,
        data.total || 0,
        data.metodo || ""
      ]);
    } 
    else if (action === "update") {
      var rowIndex = -1;
      
      // Buscar la fila por ID exacto (Mucho más rápido comparando matriz)
      for (var i = 0; i < ids.length; i++) {
        if (String(ids[i][0]) === String(data.id)) {
          rowIndex = i + 2; // +2 porque el array empieza en fila 2 (saltando encabezados)
          break;
        }
      }
      
      if (rowIndex !== -1) {
        sheet.getRange(rowIndex, 1, 1, 7).setValues([[
          data.id,
          data.fecha || sheet.getRange(rowIndex, 2).getValue(),
          data.nombre || "",
          data.producto || "Mango Con Crema 267 ml",
          data.cantidad || 0,
          data.total || 0,
          data.metodo || ""
        ]]);
      }
    } 
    else if (action === "delete") {
      var rowIndex = -1;
      
      for (var i = 0; i < ids.length; i++) {
        if (String(ids[i][0]) === String(data.id)) {
          rowIndex = i + 2;
          break;
        }
      }
      
      if (rowIndex !== -1) {
        sheet.deleteRow(rowIndex);
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
                  .setMimeType(ContentService.MimeType.JSON);
                  
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()}))
                  .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var lastRow = sheet.getLastRow();
    
    if (lastRow < 2) {
      return ContentService.createTextOutput(JSON.stringify({ "status": "success", "ventas": [] }))
                    .setMimeType(ContentService.MimeType.JSON);
    }
    
    // Traer todo el rango de datos de una sola vez (Optimización masiva de rendimiento)
    var rows = sheet.getRange(2, 1, lastRow - 1, 7).getValues();
    var ventas = [];
    
    for (var i = 0; i < rows.length; i++) {
      var row = rows[i];
      if (row[0] || row[2]) { // Validar que tenga ID o Nombre
        ventas.push({
          id: row[0],
          fecha: row[1],
          nombre: row[2],
          producto: row[3],
          cantidad: row[4],
          total: row[5],
          metodo: row[6]
        });
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({ "status": "success", "ventas": ventas }))
                  .setMimeType(ContentService.MimeType.JSON);
                  
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() }))
                  .setMimeType(ContentService.MimeType.JSON);
  }
}
